library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity traffic_light_tb is
end traffic_light_tb;

architecture Behavioral of traffic_light_tb is

    component traffic_light is
        port ( 
            clk                : in std_logic;
            reset              : in std_logic;
            rc, gc, yc, rp, gp : out std_logic
        );
    end component;

    signal clk_tb : std_logic := '0';
    signal reset_tb : std_logic := '1';
    signal rc_tb, gc_tb, yc_tb, rp_tb, gp_tb : std_logic;

    constant CLK_PERIOD : time := 125 ms; -- 8 Hz za simulaciju[cite: 2]

begin

    -- Instanciranje komponente (UUT) sa port map-om
    i1: traffic_light
        port map (
            clk   => clk_tb,
            reset => reset_tb,
            rc    => rc_tb,
            gc    => gc_tb,
            yc    => yc_tb,
            rp    => rp_tb,
            gp    => gp_tb
        );

    -- Generisanje takta preko not operatora
    clk_process : process
    begin
        clk_tb <= not clk_tb after CLK_PERIOD / 2;
        wait for CLK_PERIOD / 2;
    end process;

    -- Stimulus proces za reset i praćenje simulacije
    stim_process : process
    begin
        reset_tb <= '1';
        wait for 200 ms;

        reset_tb <= '0';

        wait for 20 sec;

        wait;
    end process;

end Behavioral;
